import math
def conv(x,w):
    if not w or len(x)<len(w): raise ValueError('invalid valid-convolution dimensions')
    return [sum(w[k]*x[n+k] for k in range(len(w))) for n in range(len(x)-len(w)+1)]
x,w,g = [1.,3.,2.,0.],[1.,-1.],[2.,-1.,3.]
y = conv(x,w)
dx,dw = [0.]*len(x),[0.]*len(w)
for n in range(len(y)):
    for k in range(len(w)):
        dx[n+k] += g[n]*w[k]
        dw[k] += g[n]*x[n+k]
def loss(x,w): return sum(a*b for a,b in zip(g,conv(x,w)))
eps=1e-6
errors=[]
for values,analytic,which in [(x,dx,'x'),(w,dw,'w')]:
    numeric=[]
    for i in range(len(values)):
        plus,minus=values.copy(),values.copy()
        plus[i]+=eps;minus[i]-=eps
        if which=='x': v=(loss(plus,w)-loss(minus,w))/(2*eps)
        else: v=(loss(x,plus)-loss(x,minus))/(2*eps)
        numeric.append(v)
    errors.extend(abs(a-b) for a,b in zip(analytic,numeric))
    print(which,'analytic',analytic,'numeric',numeric)
z=[-1.,2.,0.,3.]
lhs=conv([2*a-b for a,b in zip(x,z)],w)
rhs=[2*a-b for a,b in zip(conv(x,w),conv(z,w))]
print('output',y,'loss',loss(x,w),'max_gradient_error',max(errors))
print('linearity_error',max(abs(a-b) for a,b in zip(lhs,rhs)))
print('biased_F_zero',[v+2 for v in conv([0.]*4,w)])
assert max(errors)<1e-7
assert lhs==rhs
