import math
s = [3.,1.]
q0 = .01
def exact(t):
    return [v/(1+(v/q0-1)*math.exp(-2*v*t)) for v in s]
def run(dt):
    a, b = [math.sqrt(q0)]*2, [math.sqrt(q0)]*2
    records = {}
    for step in range(1, round(3/dt)+1):
        errors = [s[i]-a[i]*b[i] for i in range(2)]
        da = [errors[i]*b[i] for i in range(2)]
        db = [errors[i]*a[i] for i in range(2)]
        a = [a[i]+dt*da[i] for i in range(2)]
        b = [b[i]+dt*db[i] for i in range(2)]
        for time in (1,2,3):
            if step == round(time/dt): records[time] = [a[i]*b[i] for i in range(2)]
    return records
for dt in (.001,.0005):
    records = run(dt)
    for t,q in records.items():
        print('dt',dt,'t',t,'Euler',q,'exact',exact(t),
              'max_error',max(abs(a-b) for a,b in zip(q,exact(t))))
    q = records[3]
    for x in ([1,1],[1,-1]):
        pred = [q[i]*x[i] for i in range(2)]
        truth = [s[i]*x[i] for i in range(2)]
        print('input',x,'prediction',pred,'squared_error',sum((a-b)**2 for a,b in zip(pred,truth)))
