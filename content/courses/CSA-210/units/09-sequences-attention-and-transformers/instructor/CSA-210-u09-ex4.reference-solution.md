```python
import math

def softmax(scores):
    if not scores or not all(math.isfinite(x) for x in scores):
        raise ValueError('nonempty finite allowed scores required')
    maximum=max(scores)
    e=[math.exp(x-maximum) for x in scores]
    total=sum(e)
    return [x/total for x in e]

def shape(a):
    if not a or not a[0] or any(len(row)!=len(a[0]) for row in a):
        raise ValueError('nonempty rectangular matrix required')
    if not all(math.isfinite(x) for row in a for x in row):
        raise ValueError('finite entries required')
    return len(a),len(a[0])

def attention(Q,K,V,allowed=lambda i,j:True):
    nq,dk=shape(Q); nk,kk=shape(K); nv,dv=shape(V)
    if dk!=kk or nk!=nv: raise ValueError('incompatible Q/K/V dimensions')
    outputs,weights=[],[]
    for i,q in enumerate(Q):
        ids=[j for j in range(nk) if allowed(i,j)]
        if not ids: raise ValueError('attention row has no allowed key')
        scores=[sum(a*b for a,b in zip(q,K[j]))/math.sqrt(dk) for j in ids]
        p=softmax(scores)
        row=[0.]*nk
        for j,value in zip(ids,p): row[j]=value
        weights.append(row)
        outputs.append([sum(row[j]*V[j][k] for j in ids) for k in range(dv)])
    return outputs,weights

def error(a,b): return max(abs(x-y) for ra,rb in zip(a,b) for x,y in zip(ra,rb))
Q,K,V=[[math.log(3)],[0]],[[1],[0]],[[2,0],[0,4]]
full=attention(Q,K,V); causal=attention(Q,K,V,lambda i,j:j<=i)
print('full',full,'causal',causal)
assert error(full[0],[[1.5,1],[1,2]])<1e-12
assert error(causal[0],[[2,0],[1,2]])<1e-12
assert causal[1][0][1]==0
assert all(abs(sum(row)-1)<1e-12 for row in full[1]+causal[1])
scores=[1.,-2.,3.]
shift_error=max(abs(a-b) for a,b in zip(softmax(scores),softmax([x+1000 for x in scores])))
X=[[1.,0.],[0.,2.],[-1.,1.]]
p=[2,0,1]
base,_=attention(X,X,X)
permuted,_=attention([X[i] for i in p],[X[i] for i in p],[X[i] for i in p])
permutation_error=error(permuted,[base[i] for i in p])
print('score_shift_error',shift_error,'permutation_error',permutation_error)
assert shift_error<1e-12 and permutation_error<1e-12
for case in ('all_masked','dimension'):
    try:
        if case=='all_masked': attention(Q,K,V,lambda i,j:False)
        else: attention([[1.,2.]],K,V)
    except ValueError as exc: print(case,'rejected:',str(exc))
    else: raise AssertionError('invalid input was accepted')
```
